[Уровень сложности: Сложный]
Задача: "Создайте систему авторизации на основе декораторов"

[Пример решения]
def timer(func):
    import time
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"Execution time: {end-start:.2f}s")
        return result
    return wrapper
